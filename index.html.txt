    <script>
        // Floating Hearts Animation Logic
        function createHeart() {
            const heart = document.createElement('div');
            heart.classList.add('heart');
            heart.innerHTML = '❤️';
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.animationDuration = Math.random() * 3 + 2 + 's'; // 2 to 5 seconds
            document.getElementById('hearts-container').appendChild(heart);

            // Remove heart after it goes off screen to save memory
            setTimeout(() => {
                heart.remove();
            }, 5000);
        }

        // Generate a new heart every 300 milliseconds
        setInterval(createHeart, 300);

        // Transition Logic
        function revealSurprise() {
            document.getElementById('welcome-screen').style.display = 'none';
            const surprise = document.getElementById('surprise-screen');
            surprise.classList.remove('hidden');
            surprise.classList.add('fade-in');
        }
